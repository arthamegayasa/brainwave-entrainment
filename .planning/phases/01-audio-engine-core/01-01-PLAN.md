src/audio/types.ts (dibuat di plan ini):
```typescript
export type EntrainmentKind = "binaural" | "isochronic" | "monaural";
export type AmbientKind = "rain" | "ocean" | "wind" | "brown";
export type SoundKind = EntrainmentKind | "solfeggio" | AmbientKind;

export interface ToneParams {
  carrier: number; // Hz, > 0
  beat: number;    // Hz, >= 0
}

/** Satu sumber suara yang bisa di-start/stop dengan fade — diimplementasi tiap layer. */
export interface SoundLayer {
  /** Node keluaran layer; engine menyambungkan ini ke masterGain. */
  readonly output: GainNode;
  /** Mulai semua source pada waktu ctx t (detik AudioContext) dengan fade-in. */
  start(t: number): void;
  /** Fade-out lalu hentikan source; aman dipanggil sekali. */
  stop(t: number): void;
}

export type LayerFactory = (ctx: BaseAudioContext) => SoundLayer;
```
