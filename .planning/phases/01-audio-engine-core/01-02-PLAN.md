// src/audio/engine.ts
export class AudioEngine {
  constructor(ctx: BaseAudioContext);
  /** Hentikan layer aktif (fade), buat layer `kind` dari registry, start dengan fade-in. Throw Error jika kind belum terdaftar. */
  play(kind: SoundKind): void;
  /** Fade-out master lalu stop layer aktif. Aman dipanggil saat idle. */
  stop(): void;
  /** Ramp master gain ke clamp(v, 0, 1) — tidak pernah > 1.0 (safety telinga). */
  setMasterVolume(v: number): void;
  /** Daftar kind yang terdaftar di registry — App me-render tombol dari sini. */
  get implementedKinds(): SoundKind[];
}
// Registry internal: const LAYER_FACTORIES: Partial<Record<SoundKind, LayerFactory>>
